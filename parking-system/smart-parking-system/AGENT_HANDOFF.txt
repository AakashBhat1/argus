 CONTRACT  (live API surface — REAL shapes, verified against routes/ + services/)
--------------------------------------------------------------------------------
 GET    /api/stats
          -> { total, occupied, free, available, occupancy_pct, plates_today }
 GET    /api/spaces
          -> { spaces:[ {space_id, zone, floor, is_occupied, plate_text, entry_time} ],
               total, occupied, free, available, occupancy_pct, plates_today }
 POST   /api/spaces/<space_id>/release
          -> 200 { success:true, space_id, plate_text, duration_minutes, amount_paid }
          -> 404 { success:false, error }
 GET    /api/plates?limit=20            (limit clamped 1..200)
          -> [ { id, plate_text, state, timestamp, is_parked, confidence } ]
 GET    /api/plates/<int:id>
          -> { id, plate_text, state, timestamp, is_parked, exit_time,
               duration_minutes, confidence } | 404 { error }
 GET    /api/latest_plate
          -> { text, ..., profile_type, owner_name, notes }   (may be {})
 GET    /api/activity?limit=30          (limit clamped 1..200)
          -> [ { id, timestamp, event_type, description, plate_text, space_id } ]
             event_type ∈ { system, plate_detected, space_assigned, space_released,
                            vip_entry, security_alert, profile_updated, profile_deleted }
 GET    /api/session
          -> { id, start_time, end_time, plates_detected, spaces_used,
               uptime_seconds, uptime_display } | 404 { error }
 GET    /api/analytics
          -> { heatmap:{space_id:count}, predictions:[24 floats],
               actual_today:[24 ints|null], current_hour:int,
               heatmap_cells:[ {space_id, floor, zone, count} ] }   <-- floor/zone enriched
             NOTE: space_id (e.g. "G-01","F1-03") does NOT encode zone; zone (A/B/C)
             is a separate column. Consumers MUST use heatmap_cells for floor/zone,
             never parse the id. ("heatmap" dict kept for back-compat.)
 POST   /api/sandbox/entry  {plate_text, state}
          -> 200 { success:true, space_id, plate_text, state, profile_type,
                   owner_name, notes } | 400 { success:false, error }
 POST   /api/sandbox/exit   {space_id}
          -> 200 { success:true, ... } | 400 { success:false, error }
 GET    /api/billing/stats
          -> { total_revenue, total_transactions, avg_transaction,
               recent_transactions:[ {plate_text, exit_time, duration_minutes, amount_paid} ] }
 GET    /api/billing/export
          -> 200 text/csv Content-Disposition: attachment; filename="billing_export.csv"
             (returns CSV file containing plate_text, exit_time, duration_minutes, amount_paid)
 GET    /api/profiles
          -> { profiles:[ {plate_text, profile_type, owner_name, notes} ] }
 POST   /api/profiles  {plate_text, profile_type, owner_name, notes}
          -> { success:bool }   (profile_type ∈ normal|vip|blacklist)
 DELETE /api/profiles/<plate_text>
          -> { success:bool }
 POST   /api/chat  {message, mode, history}
          -> SSE text/event-stream; chunks: data:{content:"..."}; final data:{done:true}
             (mode ∈ user|admin; 400 {error} if message empty)
 GET    /api/events                      <-- PRIMARY real-time stream
          -> SSE text/event-stream (real-time push), UNNAMED data: events:
             entry:          {"event":"entry","space_id","plate_text","profile_type","owner_name","notes"}
             exit:           {"event":"exit","space_id","plate_text","duration_minutes","amount_paid"}
             profile_change: {"event":"profile_change","plate_text","profile_type","owner_name"?,"notes"?}   (owner_name+notes included on SAVE path for optimistic deltas; DELETE emits minimal {plate_text,profile_type})
             system_reset:   {"event":"system_reset","mode":"soft"|"hard"}
             config_change:  {"event":"config_change","default_spaces","rate_per_hour","total_spaces"}
             (plus ":keepalive" comment lines ~15s — EventSource swallows these)
 GET    /api/health
          -> { status:"ok", db:"ok"|"down", time:"<iso8601>" }   (always 200)
 POST   /api/session/reset   {"confirm":true, "hard"?:bool}
          -> 200 { success:true, mode:"soft"|"hard", spaces_freed:int }
          -> 400 { success:false, error }            (soft keeps history; hard wipes)
          NOTE: optional admin token gate (see below)
 POST   /api/config   {default_spaces?:int(1-200), rate_per_hour?:number(>0)}
          -> 200 { success:true, default_spaces, rate_per_hour, total_spaces }
          -> 400 { success:false, error }            (in-memory; safe space scaling)
          NOTE: optional admin token gate (see below)
 GET    /api/config
          -> { default_spaces:int, rate_per_hour:float, total_spaces:int }
          (read-only; mirrors POST success shape; for receipts + pre-fills)
 GET    /api/admin/connections
          -> { active_sse_clients:int }
          NOTE: optional admin token gate (see below)
 GET    /feed/camera , GET /feed/parking
          -> multipart/x-mixed-replace MJPEG streams (use in <img src>)
 NOTE: /session/reset, /config (POST), /admin/connections support OPTIONAL admin gating.
       When the env var ADMIN_TOKEN is set to a non-empty value on the backend, these
       three endpoints require the caller to supply header X-Admin-Token: <value>
       (preferred) or Authorization: Bearer <value>. Mismatch or missing -> 401
       {"success":false,"error":"Unauthorized"}.
       When ADMIN_TOKEN is unset/empty (the default/demo case), the endpoints pass
       through exactly as before with no 401s — this preserves compatibility with the
       current SvelteKit admin UI which does not yet send tokens. A browser-shipped
       static token is demo-grade only (not real security); real auth needs server-side
       login/session (future). The export and all other endpoints remain ungated.
 NOTE (R17): Legacy Jinja2 page routes (/, /detect, /parking, /admin) and associated
       templates/ + static/css + static/js have been retired. Flask is now a pure
       headless API server. All UI interactions occur through the SvelteKit app in web/
       (which continues to use the exact same /api + /feed contracts).
